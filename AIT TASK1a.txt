from collections import deque

n = int(input("Enter number of nodes: "))

graph = {}

for i in range(n):
    node = input(f"Enter node {i+1}: ")
    neighbours = input(
        f"Enter neighbours of {node} (space separated): "
    ).split()
    graph[node] = neighbours

start = input("Enter start node: ")
end = input("Enter end node: ")

queue = deque([[start]])
visited = set()

path = None

while queue:
    current_path = queue.popleft()
    node = current_path[-1]

    if node in visited:
        continue

    visited.add(node)

    if node == end:
        path = current_path
        break

    for neighbour in graph.get(node, []):
        if neighbour not in visited:
            queue.append(current_path + [neighbour])

if path:
    print("BFS Path:", " -> ".join(path))
else:
    print("No path found")
