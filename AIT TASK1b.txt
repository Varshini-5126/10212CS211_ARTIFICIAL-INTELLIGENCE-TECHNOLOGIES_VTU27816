n = int(input("Enter number of nodes: "))

graph = {}

for i in range(n):
    node = input(f"Enter node {i+1}: ")
    neighbours = input(
        f"Enter neighbours of {node} (space separated): "
    ).split()
    graph[node] = neighbours

start = input("Enter start node: ")
end = input("Enter end node: ")

visited = set()
path = []


def dfs(node):
    visited.add(node)
    path.append(node)

    if node == end:
        return True

    for neighbour in graph.get(node, []):
        if neighbour not in visited:
            if dfs(neighbour):
                return True

    path.pop()
    return False


if dfs(start):
    print("DFS Path:", " -> ".join(path))
else:
    print("No path found")
